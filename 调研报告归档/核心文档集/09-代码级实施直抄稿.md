### 1.4 错误码与单元失效清理

```go
// ===== internal/access/config_access/unit_mapping.go（新文件，~60 行）=====
package configaccess

import "sync"

type UnitMapping struct {
    Channel string `json:"channel,omitempty"`
    AuthRef string `json:"authRef"`
    Model   string `json:"model,omitempty"`
}

type UnitTable struct {
    mu    sync.RWMutex
    byKey map[string]UnitMapping
}

var globalUnits = &UnitTable{byKey: map[string]UnitMapping{}}

func GlobalUnits() *UnitTable { return globalUnits }

// Lookup 按 apiKey 查单元；未命中 ok=false。读多写少用 RLock。
func (t *UnitTable) Lookup(apiKey string) (UnitMapping, bool) {
    t.mu.RLock()
    defer t.mu.RUnlock()
    u, ok := t.byKey[apiKey]
    return u, ok
}

// Store 全量幂等覆盖：DSH 每次写 = 整体替换，不做增量 patch。
// units 为整批最新映射；nil/空切片 = 清空。
func (t *UnitTable) Store(units map[string]UnitMapping) {
    t.mu.Lock()
    defer t.mu.Unlock()
    t.byKey = make(map[string]UnitMapping, len(units))
    for k, v := range units {
        if k == "" || v.AuthRef == "" {
            continue // 防御：空 key / 空 authRef 不入表
        }
        t.byKey[k] = v
    }
}

// Delete 删单个 key（DSH 单单元摘除时用；通常直接 Store 全量覆盖即可）。
func (t *UnitTable) Delete(apiKey string) {
    t.mu.Lock()
    defer t.mu.Unlock()
    delete(t.byKey, apiKey)
}
```

### 2.3 关键约束：Docker 单文件挂载不能 rename（实测发现）

> `internal/api/handlers/management/config_v8.go:183-186` 现有代码注释原文：
> **"Preserve the destination inode: the standard Docker deployment mounts config.yaml as a single file, which cannot be replaced with rename."**
> 该路径最终走 `WriteConfig`（`internal/api/handlers/management/config_basic.go:101`）= `OpenFile(O_WRONLY\|O_CREATE\|O_TRUNC)` + `Write` + `f.Sync()`，**原地写、保留 inode**。

含义：对 **config.yaml（2a/2c）**，纯 rename 会在「config.yaml 被 `-v /host/config.yaml:/app/config/config.yaml` 单文件挂载」的部署下破坏 inode/挂载视图。因此 `writeFileAtomic` 不能无脑 rename，必须带**回退**：
- **auth 文件（2b）**：目录挂载，rename 安全 → 走纯 rename 原子写。
- **config.yaml（2a/2c）**：优先 rename；rename 不可用时（EXDEV 跨盘 / 单文件 bind-mount 需保 inode）**回退到原地 truncate+fsync**（等价于现有 `WriteConfig` 行为），保证 Docker 部署不回归。

### 2.4 函数签名与语义

```go
// 建议落点：internal/util/ 或 internal/config/（二者均已存在，见 ls 实测）。
// 放 internal/util/atomic_write.go，三处都能 import。

// writeFileAtomic 把 data 原子写入 path：
//  1) 在 path 同目录建临时文件（os.CreateTemp，同目录 -> 杜绝 EXDEV 跨盘 rename）；
//  2) 写入 + f.Sync() 落盘；
//  3) 关闭后 os.Rename(tmp, path) 原子替换；
//  4) 任一阶段失败 -> 删 tmp、返回 error、path 保持旧值不动、不热重载。
//
// 失败语义硬约束：绝不出现半截目标文件；失败时旧值原样保留。
func writeFileAtomic(path string, data []byte, perm os.FileMode) error
```

- **temp 同目录**：`os.CreateTemp(filepath.Dir(path), ".atomic-*")`，保证 tmp 与 target 同文件系统，`rename` 才是同盘原子操作（跨盘 rename 会 `EXDEV`）。
- **Windows rename 覆盖语义**：Go `os.Rename` 在 Windows 上底层用 `MoveFileEx(src, dst, MOVEFILE_REPLACE_EXISTING)`，**可覆盖已存在目标**（go 1.26，实测 go.mod `go 1.26.0`）；故 Windows 下无需手工先删 target。
- **跨盘**：同目录临时文件已规避；若 `filepath.Dir(path)` 与 path 不在一盘（极端），`os.CreateTemp` 在同目录仍与 path 同盘，rename 安全。
- **回退**（仅 config 用）：rename 返回错误且判定为「需保 inode / 跨盘」时，回退 `open+truncate+write+fsync` 原地写（复用 `WriteConfig` 同款，config_basic.go:101）。

```go
// ===== internal/util/atomic_write.go（新文件，~35 行）=====
package util

import (
    "os"
    "path/filepath"
)

func WriteFileAtomic(path string, data []byte, perm os.FileMode) error {
    dir := filepath.Dir(path)
    tmp, err := os.CreateTemp(dir, ".atomic-*")
    if err != nil {
        return err
    }
    tmpPath := tmp.Name()
    // 任何失败路径都兜底删 tmp
    cleanup := func() { _ = os.Remove(tmpPath) }

    if _, err := tmp.Write(data); err != nil {
        _ = tmp.Close()
        cleanup()
        return err
    }
    if err := tmp.Sync(); err != nil { // fsync 落盘，防 rename 后掉电丢数据
        _ = tmp.Close()
        cleanup()
        return err
    }
    if err := tmp.Close(); err != nil {
        cleanup()
        return err
    }
    if err := os.Rename(tmpPath, path); err != nil {
        cleanup()
        // config 场景：单文件 bind-mount 不能 rename 覆盖 -> 回退原地写保 inode
        return writeFileInPlace(path, data, perm)
    }
    return nil
}

// writeFileInPlace 原地 truncate+fsync（Docker 单文件挂载回退路径；
// 与 internal/api/handlers/management/config_basic.go:101 WriteConfig 同款语义）。
func writeFileInPlace(path string, data []byte, perm os.FileMode) error {
    f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_TRUNC, perm)
    if err != nil {
        return err
    }
    if _, err := f.Write(data); err != nil {
        _ = f.Close()
        return err
    }
    if err := f.Sync(); err != nil {
        _ = f.Close()
        return err
    }
    return f.Close()
}
```

### 2.5 「失败步不中断其余步」如何实现（真实结构）

核心 = **一个 `dead` 布尔 + 闭包 `abortDead` + 每步 `if !dead` 闸门**：

```go
// taskcenter.go:157-166
dead := false
abortDead := func(err error) bool {
    if !isGrowthSessionDead(err) {   // 仅 401/403(且非档位锁) 才算 session 死
        return false
    }
    dead = true
    add("登录态已失效（%s）——中止本轮后续步骤", err)
    return true
}
```

- **普通业务失败**（某步 error 但非 401/403 session 死）→ `abortDead` 返回 false → `add(...)` 记一行，**该步结束，进入下一步**（每步外层都包 `if !dead { ... }`，dead 仍 false，闸门放行）。例：补签失败(:193)、接单整批失败 break 本步但不置 dead(:234)、兑换失败 continue(:317)、单任务领取失败继续下一任务(:349)、抽奖失败 break 本步(:371)。
- **凭证级 401/403**（`isGrowthSessionDead` growth.go:215 = status 401/403 且 `!isGrowthTierLocked`）→ `abortDead` 置 `dead=true` → 后续所有 `if !dead` 闸门全部跳过，**不再发任何 growth 域请求**（避免堆 401）。
- **档位锁 403「连续登录天数不足」**（`isGrowthTierLocked` growth.go:225）→ **不算 session 死**，只 add「未解锁」继续（growth.go:212-221 注释：误判会因健康凭证中断整轮）。
- **永不上抛 panic**（taskcenter.go:135 注释）：所有上游失败都落进 `res.Lines`，`res.Success` 仅表「跑完」。

> ⚠DSH 直抄含义：前端按 `summary.fail` + 逐 `results[].lines` 展示即可；**不要**因单步失败就阻断整账号，也不要把 `success=true` 误读成「每步都成功」。

```ts
async function writeMinimalConfig(opts): Promise<{configPath, secretKey, probeKey}> {
  const configPath = join(opts.installRoot, 'config.yaml');   // = exe 同级目录（照 index.js:301 spawn 的 cwd）
  // 04 §4.1 步 4：写前备份
  if (existsSync(configPath)) {
    copyFileSync(configPath, `${configPath}.bak`);             // 失败不致命，仅 log
  }
  const secretKey = randomBytes(32).toString('base64url');
  const probeKey   = `sk-${randomBytes(24).toString('base64url')}`;
  const yaml = renderConfigYaml(opts, secretKey, probeKey);   // 纯函数，拼 §3.2 的 YAML 文本
  await withFileLock(configPath, async () => {
    await writeFileAtomic(configPath, yaml, { mode: 0o600, dirMode: 0o700 });
  });
  return { configPath, secretKey, probeKey };
}
```

**要点**：
1. `writeFileAtomic` 内部已做「同目录 tmp → fsync → rename」（对应 05 补丁 1 的 Go 侧同语义；TS 侧由 `dsh-atomic-write` 包封装，**不自己 `fs.writeFileSync(tmp)`**）；

**与 index.js 现状的差异（TS 化时补）**：
- 04 §4.1 要求挂 **Job Object**（父退子亡）——index.js 现状 `detached:false` + `stdio:'ignore'` **未挂 Job Object**，仅靠 `stopIfOwned()` 显式 kill（index.js:321）。Job Object 在 Windows 需 `windows-arguments` 包或调用 `kernel32.CreateJobObjectW`；⚠待核对 DSH 宿主是否已提供现成 wrapper（bridge 未用，本仓新增）。
- F5 秒崩检测：index.js 现状只靠 `waitForPort` 超时（index.js:313），**不读 exitCode**。TS 化补 `child.once('exit', (code) => { crashExitCode = code })`，若 `<startTimeoutMs` 内退出即抛 `crash`。

### 5.2 探活（三级退避 + attach 判定）

```ts
// L1 TCP：原样复用 index.js:132 probePort
async function healthProbe(port, deadlineMs, secretKey) {
  const schedule = [0, 100, 300, 1000, 2000];   // 之后固定 2s（04 §3.4）
  const deadline = Date.now() + deadlineMs;
  let i = 0;
  for (;;) {
    const tcp = await probePort(port, 1200);
    let healthz = false, readyz = false, handshake = false;
    if (tcp) {
      healthz  = await tryGet(`http://127.0.0.1:${port}/healthz`, { ok: 200 });
      // /readyz 依赖 05 补丁 2；未打补丁时 404/503 一律记 false 但不致命
      readyz   = await tryGet(`http://127.0.0.1:${port}/readyz`, { ok: 200 });
      if (healthz) {
        handshake = await tryGet(
          `http://127.0.0.1:${port}/v0/management/config`,
          { ok: 200, headers: { authorization: `Bearer ${secretKey}` } },
        );
      }
    }
    if (tcp && healthz && readyz && handshake) return { tcp, healthz, readyz, handshake };
    if (Date.now() > deadline) throw new BootstrapError('start-timeout', ...);
    await sleep(schedule[Math.min(i++, schedule.length - 1)]);
  }
}
```

### 5.3 attach 已在跑实例的判定（F4）

照 04 §4.2 + index.js:277 实测：**进 S4 spawn 之前先 `probePort(port)` + `GET /healthz`**；通则不 spawn，直接返回 `{ child: undefined, owned: false, attached: true }`。无锁文件，探活是唯一判据。
- 注意：attach 分支的 `secretKey` 是**已在跑实例**的 key，bootstrap 无法知道 → 从插件 schema `adminKey` / `adminKeyRef`（`index.js:234 resolveAdminKey`）读，不能用首启新生成的那把。

## 专题 5：直选链路 adapter 最小可跑骨架（TS）

### 5.0 设计映射：bridge 七环 → 本插件直选链路

bridge 的形态是「一个 variant = 一个 provider = 一个 shim = 一个 adapter」。本插件（cpa-panel 直选）的形态是「CPA 已经是 OpenAI 兼容端点（`http://127.0.0.1:8317/v1`），不需要 shim，不需要 OAuth store，直接把每个账号 unit 注册成一个 adapter」。七环逐环对应：

| 环 | bridge 做什么 | 本骨架改成什么 |
|---|---|---|
| ① inject | `export const inject = ['llm']` | 原样照抄 |
| ② 构造 adapter | `createWorkBuddyAdapter({ providerId, displayName, shim, store, catalog, ... })` | 砍掉 shim/store/catalog/probe/hidden，改为 `createCpaUnitAdapter({ unit, baseUrl })` |
| ③ invalidate+emit | `runtime.invalidate = () => { workbuddy.invalidate(); ctx.emit('llm/adapters-updated') }` | 原样照抄，units 列表变更时触发 |
| ④ 不注册目录项 | 注释明确"contributes NO registerConfigurableProviders directory entry" | 原样照抄——直选账号不出现在 Models 设置页 |
| ⑤ registerAdapter | `ctx.llm.registerAdapter([variant.id], workbuddy.adapter)` | 每个 unit 调一次，providerId = `cpa-<authRef>` |
| ⑥ releaseAdapter 清理 | `ctx.effect(() => () => { releaseAdapter(); void shim.close() })` | 去掉 `shim.close()`（无 shim），只留 `releaseAdapter()` |
| ⑦ PiAiAdapter 组装 | `createProvider` → `ResolvedPiAiProviderProfile` → `new PiAiAdapter({ profiles, auth: INERT_AUTH, resolveApiKey })` | 原样照抄结构，`resolveApiKey` 闭包到 `units[i].apiKey`，`baseUrl` 直接写死 `http://127.0.0.1:8317/v1` |

### 5.1 数据形状（替换业务字段）

### 5.4 reasoningEffort 注入点（与 M4 一致：adapter 构造请求处）

题目要求：客户端已传 → 不动；客户端未传 → 按 `units[].reasoningEffortDefault` 兜底，全局默认 `medium`。

- **候选 A（推荐，若 seam 支持）**：override `resolveModel`，在返回的 `LlmResolvedModelInfo` 上挂默认 effort。bridge `WorkBuddyPiAiAdapter.resolveModel`（`:355-364`）已经 override 了此方法改 display name，照抄加一行：

- **候选 B（兜底）**：包装 `openAICompletionsApi()`，在 request body 构造处注入：

### 5.5 units 变更时 re-register 的时机

bridge 用 `CREDENTIAL_POLL_MS = 30_000`（`:197`）轮询 credential 文件存在性 + `runtime.invalidate()`（`:683-686`）让单个 adapter 内部 profiles 重算。本插件 units 来自面板半边 HTTP 配置，不需要文件轮询，re-register 时机改为：

1. **启动时**：`registerAll()` 全量注册一次（见 5.2）。
2. **面板半边 POST /internal/units-changed**：浏览器半边增删账号 → 面板半边转发 → 宿主半边读新 units → `registerAll()` 整体重建（释放旧 releases → 重新 `ctx.llm.registerAdapter` → `ctx.emit('llm/adapters-updated')`）。
3. **不做文件轮询**：units 不落地成文件，纯内存 + 面板半边 HTTP，避免 bridge 那套 30s poll 的复杂度。

> **取舍**：整体 re-register 比 bridge 的单 adapter invalidate 粗，但 unit 数量小（个位数），且直选链路没有 catalog 刷新/probe 重试的负担，整体重建足够。

### 6.2 Job Object 父退子亡（Windows）

Node 侧**无原生 Job Object API**。三条实现路径：

| 路径 | 做法 | 推荐度 |
|---|---|---|
| **A. npm 包 `node-windows-job` / `windoze` 类** | spawn 后用 Win32 API `AssignProcessToJobObject` 把子进程绑到本进程的 Job，Job 设 `KILL_ON_JOB_CLOSE`。父进程死 → OS 自动杀 Job 内所有进程树。 | ⚠待核对-5：包存在性、维护状态、是否支持子进程再 fork（CPA 会不会再 spawn 子进程？） |
| **B. PowerShell 包装** | 父进程启动一个 PowerShell 监听器，监听父 PID 存活，死则 `taskkill /PID <childPid> /T /F`。 | 备选：多一层 PowerShell 进程，多一个故障点 |
| **C. taskkill /T 兜底（最朴素）** | 父进程 `process.on('exit' / 'beforeExit' / 'SIGINT' / 'SIGTERM')` 时调 `taskkill /PID <pid> /T /F`。 | **推荐作为基线**：零依赖、行为可预期；缺点是父进程被 SIGKILL（如调试器强杀）时 exit handler 不跑，兜底失效 |

> **推荐**：基线走 **C**（零依赖先跑通），若现场验证 DSH 宿主被强杀时 CPA 残留，再叠加 **A**（`node-windows-job`）作为硬保证。B 不推荐。
>
> **实现细节待实施时验证**：`taskkill /T` 是否覆盖 CPA 自己 fork 出来的子进程（如 node runtime 子进程）；`/F` 是否会让 CPA 来不及 flush 配置（05 设计规格补丁 1 已要求三处写盘改 atomic，故 /F 安全）。

`detached` 取舍：cpa-panel 用 `detached: false` + `unref()`。在 Job Object 路线下，`detached: false` 让子进程跟父进程同控制台组，配合 `windowsHide:true` 不弹窗；`unref()` 让父进程不被子进程的 stdout 阻塞。保持 cpa-panel 现状，**不改**。

### 6.3 退避重启（指数退避 + jitter + 时间窗上限）

cpa-panel **完全没有**重启逻辑。本骨架新增：

```ts
interface BackoffPolicy {
  initialDelayMs: number     // 默认 1_000
  multiplier: number         // 默认 2.0
  maxDelayMs: number         // 时间窗上限，默认 30_000
  jitterRatio: number       // 0~1，默认 0.3（±30% 抖动）
  maxRestartsPerWindow: number  // 默认 5（窗口内最多重启次数，防爆）
  windowMs: number           // 默认 60_000（统计窗口）
}

// 伪码：
let restartCountInWindow = 0
let windowStart = Date.now()

function onChildExit(code: number | null, signal: NodeJS.Signals | null) {
  emit(signal === null ? 'exited' : 'crashed', { code, signal })

  // 窗口计数
  if (Date.now() - windowStart > windowMs) { windowStart = Date.now(); restartCountInWindow = 0 }
  if (restartCountInWindow >= maxRestartsPerWindow) {
    emit('give-up', { reason: 'too-many-restarts' })
    return
  }
  restartCountInWindow++

  const delay = Math.min(
    initialDelayMs * (multiplier ** (restartCountInWindow - 1)),
    maxDelayMs,
  )
  const jitter = delay * jitterRatio * (Math.random() * 2 - 1)
  setTimeout(() => start(), delay + jitter)
}
```

```ts
  shutdownEndpoint: string      // 'http://127.0.0.1:8317/v8/management/shutdown'
  managementKey: string
  gracefulTimeoutMs: number     // 默认 5_000
  forceTimeoutMs: number       // 默认 3_000
}

export type CpaProcessState =
  | 'starting'
  | 'running'
  | 'crashed'      // 子进程退出但本插件仍在托管（等待退避重启）
  | 'exited'       // 子进程退出且本插件不再重启（give-up 或主动 stop）

export interface CpaProcessEvents {
  'state': (state: CpaProcessState, detail?: { code?: number; signal?: string; reason?: string }) => void
  'ready': () => void           // 端口探活通过
  'restart': (attempt: number, nextDelayMs: number) => void
}

export interface CpaProcessHost {
  start(params: CpaStartParams): Promise<{ running: boolean; reason?: string }>
  stop(params: CpaStopParams): Promise<void>
  state(): CpaProcessState
  on<K extends keyof CpaProcessEvents>(event: K, handler: CpaProcessEvents[K]): void
}

// 健康探活接线（照 cpa-panel :132-157 直抄）：
//   - start() 里 spawn 后调 waitForPort(port, startTimeoutMs) → 通过则 emit 'ready' + state='running'
//   - child.on('exit') → 走 6.3 退避逻辑
//   - child.on('error') → 同 exit 处理
```
